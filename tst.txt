int findMax(int arr[], int n) {
    if (n <= 0) return 0;
    int maxVal = arr[0];
    for (int i = 1; i < n; i++) {
        if (arr[i] > maxVal) {

Linked List: Insert at End

Write a function 'Node* insertEnd(Node* head, int value)' that adds a new node with the given value to the end of a singly linked list. The function must return the head of the list. Ensure the new node is properly initialized.

struct Node {
    int data;
    Node* next;
};

Node* insertEnd(Node* head, int val) {
    Node* newNode = new Node{val, nullptr};
    if (head == nullptr) return newNode;
    
    Node* temp = head;
    while (temp->next != nullptr) {
        temp = temp->next;
    }
    temp->next = newNode;
    return head;
}
